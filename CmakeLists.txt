cmake_minimum_required(VERSION 3.15)
project(PixelRender)

set(CMAKE_CXX_STANDARD 17)

# Diretórios
set(SRC_DIR src)
set(EXTERNAL_DIR external)
set(IMGUI_DIR ${EXTERNAL_DIR}/imgui)
set(RLIMGUI_DIR ${EXTERNAL_DIR}/rlimgui)
set(IMGUIFILEDIALOG_DIR ${EXTERNAL_DIR}/imguifiledialog)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/out)


# Arquivos do seu jogo
file(GLOB GAME_SOURCES
    ${SRC_DIR}/*.cpp
)
file(GLOB IMGUI_SRC
    ${IMGUI_DIR}/*.cpp
)
file(GLOB RLIMGUI_SRC
    ${RLIMGUI_DIR}/*.cpp
)
file(GLOB IMGUIFILEDIALOG_SRC
    ${IMGUIFILEDIALOG_DIR}/*.cpp
)

# ImGui (compilado como biblioteca própria)
add_library(imgui STATIC
    ${IMGUI_SRC}
)

target_include_directories(imgui PUBLIC
    ${IMGUI_DIR}/include
)

# rlImGui (biblioteca separada)
add_library(rlimgui STATIC
    ${RLIMGUI_SRC}
)

target_include_directories(rlimgui PUBLIC
    ${IMGUI_DIR}/include
    ${RLIMGUI_DIR}/include
    include
)
# rlImGui (biblioteca separada)
add_library(imguifiledialog STATIC
    ${IMGUIFILEDIALOG_SRC}
)

target_include_directories(imguifiledialog PUBLIC
    ${IMGUI_DIR}/include
    ${IMGUIFILEDIALOG_DIR}/include
)

# Binário final do jogo
add_executable(game
    ${GAME_SOURCES}
)
set_target_properties(${PixelRender} PROPERTIES OUTPUT_NAME "Pixel Render")
# Raylib
# Se estiver usando MinGW com .a dentro de lib/
target_link_directories(game PRIVATE lib)

target_include_directories(game PRIVATE
    include
    ${IMGUI_DIR}/include
    ${RLIMGUI_DIR}/include
    ${IMGUIFILEDIALOG_DIR}/include
)
